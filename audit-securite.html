Critique 1. Exécution arbitraire de fichiers via IPC non validé (`shell.openPath`) CWE-78 / CWE-22
✓ Corrigé
Fichier : src/main/main.js:1645-1668 Catégorie OWASP : A01:2021 Broken Access Control / A03 Injection Impact : Exécution de binaires locaux ou UNC via IPC

Les canaux IPC download:open-file et download:show-folder transmettaient directement l'argument filePath à shell.openPath() sans vérifier l'appartenance au dossier Downloads ni bloquer les chemins réseau UNC (\\attacker\share).

❌ Avant (Vulnérable)src/main/main.js
ipcMain.on('download:open-file', (e, filePath) => {
  if (filePath) shell.openPath(filePath);
});
ipcMain.on('download:show-folder', (e, filePath) => {
  if (filePath) shell.showItemInFolder(filePath);
});
✅ Après (Corrigé)src/main/main.js
function isSafeDownloadPath(targetPath) {
  if (!targetPath || typeof targetPath !== 'string') return false;
  if (targetPath.startsWith('\\\\')) return false;
  const resolved = path.resolve(targetPath);
  const downloadsDir = path.resolve(app.getPath('downloads'));
  const inDownloadsDir = resolved.toLowerCase().startsWith(downloadsDir.toLowerCase() + path.sep);
  const inRecordedList = downloads.some(d => d.savePath && path.resolve(d.savePath) === resolved);
  return (inDownloadsDir || inRecordedList) && fs.existsSync(resolved);
}
ipcMain.on('download:open-file', (e, filePath) => {
  if (isSafeDownloadPath(filePath)) shell.openPath(path.resolve(filePath));
});

✅ Après (Corrigé)src/main/store.js
encryptSecret(plainText) {
  if (!plainText || plainText.startsWith('dpapi:')) return plainText;
  if (safeStorage && safeStorage.isEncryptionAvailable()) {
    return 'dpapi:' + safeStorage.encryptString(plainText).toString('base64');
  }
  return plainText;
}
verifyVaultPin(pin) {
  const candidate = crypto.pbkdf2Sync(String(pin).trim(), this.data.vaultPinSalt, 100000, 32, 'sha256');
  const stored = Buffer.from(this.data.vaultPinHash, 'hex');
  return candidate.length === stored.length && crypto.timingSafeEqual(candidate, stored);
}
